function aqnl_installa() {
    if (get_option('aqnl_db_versione') === '1') { return; }
    global $wpdb;
    $t = aqnl_tabella();
    $c = $wpdb->get_charset_collate();
    require_once ABSPATH . 'wp-admin/includes/upgrade.php';
    dbDelta("CREATE TABLE $t (
  id bigint(20) unsigned NOT NULL AUTO_INCREMENT,
  email varchar(190) NOT NULL,
  nome varchar(120) NOT NULL DEFAULT '',
  citta varchar(120) NOT NULL DEFAULT '',
  lista varchar(30) NOT NULL DEFAULT 'pubblico',
  stato varchar(20) NOT NULL DEFAULT 'in_attesa',
  token varchar(64) NOT NULL DEFAULT '',
  fonte varchar(190) NOT NULL DEFAULT '',
  consenso text NULL,
  creato datetime NULL,
  confermato datetime NULL,
  disiscritto datetime NULL,
  PRIMARY KEY  (id),
  UNIQUE KEY email (email)
) $c;");
    if (!get_option('aqnl_chiave_api')) { add_option('aqnl_chiave_api', wp_generate_password(40, false, false), '', 'no'); }
    if (!get_option('aqnl_chiave_hmac')) { add_option('aqnl_chiave_hmac', wp_generate_password(48, false, false), '', 'no'); }
    update_option('aqnl_db_versione', '1', false);
}
add_action('init', 'aqnl_installa');

function aqnl_norm($e) { return strtolower(trim((string) $e)); }

function aqnl_token_disiscrizione($email) {
    return substr(hash_hmac('sha256', 'u|' . aqnl_norm($email), (string) get_option('aqnl_chiave_hmac')), 0, 32);
}

function aqnl_url_pagina($args = array()) {
    $base = home_url('/newsletter/');
    return $args ? add_query_arg($args, $base) : $base;
}

function aqnl_url_disiscrizione($email) {
    return add_query_arg(array('e' => rawurlencode(aqnl_norm($email)), 't' => aqnl_token_disiscrizione($email)),
        rest_url('aquilante/v1/disiscrizione'));
}

function aqnl_testo($chiave, $predefinito) {
    $v = get_option('aqnl_testo_' . $chiave);
    return (is_string($v) && $v !== '') ? $v : $predefinito;
}

function aqnl_mail($a, $oggetto, $corpo) {
    $intestazioni = array(
        'From: Giulio Cavalli, Teatro Aquilante <info@teatroaquilante.it>',
        'Reply-To: info@teatroaquilante.it',
        'Content-Type: text/plain; charset=UTF-8',
    );
    return wp_mail($a, $oggetto, $corpo, $intestazioni);
}

function aqnl_sostituisci($testo, $riga) {
    $nome = trim((string) $riga['nome']);
    $citta = trim((string) $riga['citta']);
    $riga_citta = $citta !== ''
        ? 'Ci leggi da ' . $citta . ': quando una data cade dalle tue parti, te lo scrivo in cima alla lettera.'
        : 'Una cosa ti chiedo: rispondi a questa mail e dimmi da dove scrivi. Se passiamo dalle tue parti, lo saprai prima degli altri.';
    return strtr($testo, array(
        '{riga_citta}' => $riga_citta,
        '{saluto}' => $nome !== '' ? 'Ciao ' . $nome . ',' : 'Ciao,',
        '{link_conferma}' => add_query_arg(array('e' => rawurlencode($riga['email']), 't' => $riga['token']), rest_url('aquilante/v1/conferma')),
        '{link_disiscrizione}' => aqnl_url_disiscrizione($riga['email']),
        '{link_date}' => home_url('/appuntamenti/'),
        '{link_produzioni}' => home_url('/spettacoli/'),
    ));
}

function aqnl_ip_hash() {
    $ip = isset($_SERVER['REMOTE_ADDR']) ? (string) $_SERVER['REMOTE_ADDR'] : '';
    return substr(hash('sha256', $ip . '|' . (string) get_option('aqnl_chiave_hmac')), 0, 16);
}

function aqnl_risposta($request, $ok, $esito, $codice = 200) {
    $vuole_json = stripos((string) $request->get_header('accept'), 'application/json') !== false
        || $request->get_header('x_requested_with') || $request->get_param('ajax');
    if ($vuole_json) {
        return new WP_REST_Response(array('ok' => $ok, 'esito' => $esito), $codice);
    }
    $r = new WP_REST_Response(null, 303);
    $r->header('Location', aqnl_url_pagina(array('esito' => $esito)));
    return $r;
}

function aqnl_rest_iscrizione($request) {
    global $wpdb;
    $t = aqnl_tabella();
    if ((string) $request->get_param('sito_web') !== '') { return aqnl_risposta($request, true, 'controlla-posta'); }
    $email = aqnl_norm($request->get_param('email'));
    if (!is_email($email) || strlen($email) > 190) { return aqnl_risposta($request, false, 'email-non-valida', 400); }
    if ((string) $request->get_param('privacy') !== '1') { return aqnl_risposta($request, false, 'privacy', 400); }
    $chiave_rl = 'aqnl_rl_' . aqnl_ip_hash();
    $n = (int) get_transient($chiave_rl);
    if ($n >= 6) { return aqnl_risposta($request, false, 'troppi-tentativi', 429); }
    set_transient($chiave_rl, $n + 1, HOUR_IN_SECONDS);
    $nome = mb_substr(sanitize_text_field((string) $request->get_param('nome')), 0, 120);
    $citta = mb_substr(sanitize_text_field((string) $request->get_param('citta')), 0, 120);
    $fonte = mb_substr(esc_url_raw((string) $request->get_param('fonte')), 0, 190);
    $consenso = 'Consenso privacy newsletter dato il ' . current_time('mysql') . ' dal modulo ' . $fonte . ' (rete ' . aqnl_ip_hash() . ')';
    $riga = $wpdb->get_row($wpdb->prepare("SELECT * FROM $t WHERE email = %s", $email), ARRAY_A);
    if ($riga && $riga['stato'] === 'attivo') { return aqnl_risposta($request, true, 'gia-iscritto'); }
    $token = wp_generate_password(32, false, false);
    $dati = array('nome' => $nome, 'citta' => $citta, 'lista' => 'pubblico', 'stato' => 'in_attesa', 'token' => $token,
        'fonte' => $fonte, 'consenso' => $consenso, 'creato' => current_time('mysql'), 'disiscritto' => null);
    if ($riga) {
        $wpdb->update($t, $dati, array('id' => $riga['id']));
    } else {
        $dati['email'] = $email;
        $wpdb->insert($t, $dati);
    }
    $riga = $wpdb->get_row($wpdb->prepare("SELECT * FROM $t WHERE email = %s", $email), ARRAY_A);
    aqnl_mail($email, aqnl_testo('conferma_oggetto', AQNL_CONFERMA_OGGETTO), aqnl_sostituisci(aqnl_testo('conferma_corpo', AQNL_CONFERMA_CORPO), $riga));
    return aqnl_risposta($request, true, 'controlla-posta');
}

function aqnl_rest_conferma($request) {
    global $wpdb;
    $t = aqnl_tabella();
    $email = aqnl_norm($request->get_param('e'));
    $tok = (string) $request->get_param('t');
    $riga = $wpdb->get_row($wpdb->prepare("SELECT * FROM $t WHERE email = %s", $email), ARRAY_A);
    $esito = 'link-scaduto';
    if ($riga && $tok !== '' && hash_equals((string) $riga['token'], $tok)) {
        if ($riga['stato'] !== 'attivo') {
            $wpdb->update($t, array('stato' => 'attivo', 'confermato' => current_time('mysql'), 'disiscritto' => null), array('id' => $riga['id']));
            aqnl_mail($email, aqnl_testo('benvenuto_oggetto', AQNL_BENVENUTO_OGGETTO), aqnl_sostituisci(aqnl_testo('benvenuto_corpo', AQNL_BENVENUTO_CORPO), $riga));
        }
        $esito = 'confermata';
    }
    $r = new WP_REST_Response(null, 303);
    $r->header('Location', aqnl_url_pagina(array('esito' => $esito)));
    return $r;
}

function aqnl_segna_disiscritto($email) {
    global $wpdb;
    $t = aqnl_tabella();
    $riga = $wpdb->get_row($wpdb->prepare("SELECT id FROM $t WHERE email = %s", $email), ARRAY_A);
    if ($riga) {
        $wpdb->update($t, array('stato' => 'disiscritto', 'disiscritto' => current_time('mysql')), array('id' => $riga['id']));
    } else {
        $wpdb->insert($t, array('email' => $email, 'lista' => 'esterno', 'stato' => 'disiscritto', 'creato' => current_time('mysql'), 'disiscritto' => current_time('mysql')));
    }
}

function aqnl_rest_disiscrizione($request) {
    $email = aqnl_norm($request->get_param('e'));
    $tok = (string) $request->get_param('t');
    $valido = is_email($email) && $tok !== '' && hash_equals(aqnl_token_disiscrizione($email), $tok);
    if ($request->get_method() === 'GET') {
        $r = new WP_REST_Response(null, 303);
        $r->header('Location', $valido ? aqnl_url_pagina(array('disiscrivi' => '1', 'e' => rawurlencode($email), 't' => $tok)) : aqnl_url_pagina(array('esito' => 'link-scaduto')));
        return $r;
    }
    if (!$valido) { return aqnl_risposta($request, false, 'link-scaduto', 400); }
    aqnl_segna_disiscritto($email);
    if ((string) $request->get_param('List-Unsubscribe') === 'One-Click') {
        return new WP_REST_Response(array('ok' => true), 200);
    }
    return aqnl_risposta($request, true, 'disiscritto');
}

function aqnl_chiave_ok($request) {
    $k = (string) $request->get_header('x_aq_key');
    $vera = (string) get_option('aqnl_chiave_api');
    return $vera !== '' && $k !== '' && hash_equals($vera, $k);
}

function aqnl_rest_citta($request) {
    global $wpdb;
    $t = aqnl_tabella();
    $email = aqnl_norm($request->get_param('email'));
    $citta = mb_substr(sanitize_text_field((string) $request->get_param('citta')), 0, 120);
    $riga = $wpdb->get_row($wpdb->prepare("SELECT id, stato FROM $t WHERE email = %s", $email), ARRAY_A);
    if (!$riga) { return new WP_REST_Response(array('ok' => false, 'esito' => 'non-iscritto'), 404); }
    $wpdb->update($t, array('citta' => $citta), array('id' => $riga['id']));
    return new WP_REST_Response(array('ok' => true, 'esito' => 'aggiornata', 'stato' => $riga['stato']), 200);
}

function aqnl_rest_iscritti($request) {
    global $wpdb;
    $t = aqnl_tabella();
    $righe = $wpdb->get_results("SELECT email, nome, citta, lista, stato, fonte, consenso, creato, confermato, disiscritto FROM $t ORDER BY id", ARRAY_A);
    $r = new WP_REST_Response(array('ok' => true, 'quando' => current_time('mysql'), 'righe' => $righe), 200);
    $r->header('Cache-Control', 'no-store');
    return $r;
}

function aqnl_rest_chiavi($request) {
    return array('api' => get_option('aqnl_chiave_api'), 'hmac' => get_option('aqnl_chiave_hmac'));
}

function aqnl_rest_testi($request) {
    $ammessi = array('conferma_oggetto', 'conferma_corpo', 'benvenuto_oggetto', 'benvenuto_corpo');
    $fatti = array();
    foreach ($ammessi as $k) {
        $v = $request->get_param($k);
        if (is_string($v)) { update_option('aqnl_testo_' . $k, $v, false); $fatti[] = $k; }
    }
    $out = array('aggiornati' => $fatti);
    foreach ($ammessi as $k) { $out[$k] = aqnl_testo($k, constant('AQNL_' . strtoupper($k))); }
    return $out;
}

function aqnl_righe_admin($stato = '') {
    global $wpdb;
    $t = aqnl_tabella();
    if ($stato !== '') {
        return $wpdb->get_results($wpdb->prepare("SELECT * FROM $t WHERE stato = %s ORDER BY id DESC", $stato), ARRAY_A);
    }
    return $wpdb->get_results("SELECT * FROM $t ORDER BY id DESC", ARRAY_A);
}

function aqnl_pagina_admin() {
    if (!current_user_can('manage_options')) { return; }
    global $wpdb;
    $t = aqnl_tabella();
    $stati = array('' => 'Tutti', 'attivo' => 'Iscritti', 'in_attesa' => 'In attesa di conferma', 'disiscritto' => 'Usciti dalla lista');
    $stato = isset($_GET['stato']) && isset($stati[$_GET['stato']]) ? sanitize_key($_GET['stato']) : '';
    $conti = $wpdb->get_results("SELECT stato, COUNT(*) n FROM $t GROUP BY stato", OBJECT_K);
    $righe = aqnl_righe_admin($stato);
    $csv = wp_nonce_url(admin_url('admin-post.php?action=aqnl_csv&stato=' . $stato), 'aqnl_csv');
    echo '<div class="wrap"><h1>Iscritti alla lettera mensile "Dove siamo"</h1>';
    echo '<p>Persone iscritte dal sito (teatroaquilante.it/newsletter) e indirizzi usciti dalla lista con il link in fondo alle lettere. I programmatori e i lavoratori della compagnia stanno nei file del progetto e qui compaiono solo se si sono disiscritti.</p>';
    echo '<ul class="subsubsub">';
    $voci = array();
    foreach ($stati as $k => $etichetta) {
        $n = $k === '' ? array_sum(array_map(function ($o) { return (int) $o->n; }, $conti)) : (isset($conti[$k]) ? (int) $conti[$k]->n : 0);
        $url = admin_url('tools.php?page=aqnl-iscritti' . ($k !== '' ? '&stato=' . $k : ''));
        $voci[] = '<li><a href="' . esc_url($url) . '"' . ($k === $stato ? ' class="current"' : '') . '>' . esc_html($etichetta) . ' <span class="count">(' . $n . ')</span></a>';
    }
    echo implode(' | </li>', $voci) . '</li></ul>';
    echo '<p style="clear:both;padding-top:8px"><a class="button button-primary" href="' . esc_url($csv) . '">Scarica per Excel</a></p>';
    echo '<table class="widefat striped"><thead><tr><th>Email</th><th>Nome</th><th>Città</th><th>Lista</th><th>Stato</th><th>Iscritto il</th><th>Confermato il</th><th>Uscito il</th><th>Da dove</th></tr></thead><tbody>';
    if (!$righe) { echo '<tr><td colspan="9">Nessuna riga.</td></tr>'; }
    foreach ($righe as $r) {
        echo '<tr><td>' . esc_html($r['email']) . '</td><td>' . esc_html($r['nome']) . '</td><td>' . esc_html($r['citta']) . '</td><td>' . esc_html($r['lista']) . '</td><td>' . esc_html($stati[$r['stato']] ?? $r['stato']) . '</td><td>' . esc_html((string) $r['creato']) . '</td><td>' . esc_html((string) $r['confermato']) . '</td><td>' . esc_html((string) $r['disiscritto']) . '</td><td>' . esc_html($r['fonte']) . '</td></tr>';
    }
    echo '</tbody></table></div>';
}

function aqnl_scarica_csv() {
    if (!current_user_can('manage_options')) { wp_die('Permesso negato'); }
    check_admin_referer('aqnl_csv');
    $stato = isset($_GET['stato']) ? sanitize_key($_GET['stato']) : '';
    $righe = aqnl_righe_admin($stato);
    nocache_headers();
    header('Content-Type: text/csv; charset=UTF-8');
    header('Content-Disposition: attachment; filename="iscritti-newsletter-' . gmdate('Y-m-d') . '.csv"');
    $out = fopen('php://output', 'w');
    fwrite($out, "\xEF\xBB\xBF");
    fputcsv($out, array('Email', 'Nome', 'Città', 'Lista', 'Stato', 'Iscritto il', 'Confermato il', 'Uscito il', 'Da dove', 'Consenso'), ';');
    foreach ($righe as $r) {
        fputcsv($out, array($r['email'], $r['nome'], $r['citta'], $r['lista'], $r['stato'], $r['creato'], $r['confermato'], $r['disiscritto'], $r['fonte'], $r['consenso']), ';');
    }
    fclose($out);
    exit;
}
add_action('admin_post_aqnl_csv', 'aqnl_scarica_csv');
